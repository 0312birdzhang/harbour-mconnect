# Keep the existing bundle paths available through relative links.
function(install_relative_link name target)
  install(CODE "
    execute_process(COMMAND \"${CMAKE_COMMAND}\" -E create_symlink \"${target}\"
      \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${name}\"
      RESULT_VARIABLE link_result)
    if(NOT link_result EQUAL 0)
      message(FATAL_ERROR \"Cannot install the Mconnect link ${name}\")
    endif()
  ")
endfunction()
install_relative_link(bin/harbour-mconnect harbour-mconnect-silica)

install_relative_link("${bundle}/mconnect-silica" "../../${native_bundle}/mconnect-silica")

install_relative_link("${bundle}/lib/libmconnect_audio.so" "../../../${native_bundle}/lib/libmconnect_audio.so")
